## Teknisk rekommendation
```python
# Pseudokod för minneslagring
def store_agent_memory(agent_id, event_data):
    # Generera relevansscore baserat på agentens intressen
    relevance = calculate_relevance(agent_id, event_data)

    # Skapa kontexttaggar
    tags = generate_context_tags(event_data)

    # Lagra i Supabase
    supabase.from_('agent_memory').insert({
        agent_id: agent_id,
        event_id: uuid4(),
        timestamp: datetime.now(),
        content: event_data['content'],
        relevance_score: relevance,
        context_tags: tags
    }).execute()

# Exempel på återkallning
def recall_memory(agent_id, query):
    results = supabase.from_('agent_memory')
        .select('*')
        .eq('agent_id', agent_id)
        .ilike('content', f'%{query}%')
        .order('relevance_score', {desc: True})
        .limit(5)
        .execute()

    return results.data
```
