# Pseudokod för agent_memory.db
CREATE TABLE agent_memory (
    agent_id TEXT,
    event_type TEXT,  # 'decision', 'vote', 'transaction'
    event_data JSON,  # specifik data om händelsen
    timestamp DATETIME,
    outcome_impact FLOAT,  # -1.0 till 1.0 för konsekvens
    PRIMARY KEY (agent_id, timestamp)
)

# Exempel på minneslagring
def record_decision(agent, decision_type, outcome):
    db.execute("""
        INSERT INTO agent_memory
        VALUES (?, ?, ?, ?, ?)
    """, (agent.id, 'decision', decision_type, datetime.now(), outcome))

# Exempel på minnesanvändning
