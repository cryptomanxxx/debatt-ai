```python
# Pseudokod för minnesmekanism
class AgentMemory:
    def __init__(self, agent_id):
        self.agent_id = agent_id
        self.memory_store = {}  # {timestamp: {"event": str, "context": dict}}

    def record_event(self, event_type, details, related_agents=None):
        timestamp = datetime.now()
        self.memory_store[timestamp] = {
            "event": event_type,
            "details": details,
            "related_agents": related_agents or [],
            "context": self._generate_context()
        }

    def recall_events(self, filters=None):
        """Returnerar relevanta händelser baserat på filter"""
        if not filters:
            return list(self.memory_store.values())

        results = []
        for event in self.memory_store.values():
            if all(event["details"].get(k) == v for k, v in filters.items()):
                results.append(event)
        return results

    def _generate_context(self):
        """Genererar kontextuell metadata om aktuell situation"""
        return {
            "current_coalitions": get_active_coalitions(self.agent_id),
            "recent_votes": get_recent_votes(self.agent_id),
            "economic_status": get_economic_status(self.agent_id)
        }
```
